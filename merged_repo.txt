def log(msg: str):
    ts = datetime.now().isoformat(timespec="seconds")
    LOG.parent.mkdir(parents=True, exist_ok=True)
    with LOG.open("a", encoding="utf-8") as f:
        f.write(f"[{ts}] {msg}\n")


def load_cfg():
    return yaml.safe_load((ROOT / "config.yaml").read_text(encoding="utf-8"))


def read_features_csv(cfg):
    p = Path(cfg["files"]["features_csv"])
    if not p.exists():
        raise FileNotFoundError(f"features_csv not found: {p}")
    df = pd.read_csv(p, parse_dates=["time"])
    return df.sort_values("time").reset_index(drop=True)


def build_target(df: pd.DataFrame, price_col="close", horizon=1):
    """Target = jednokrokowa przyszła zmiana procentowa."""
    return df[price_col].pct_change().shift(-horizon).rename("target_fwd1")


def add_lags(df: pd.DataFrame, cols: list, max_lag: int) -> pd.DataFrame:
    """Dodaje lagi 1..max_lag dla listy kolumn."""
    if max_lag <= 0 or not cols:
        return df
    lag_blocks = []
    for L in range(1, max_lag + 1):
        blk = df[cols].shift(L)
        blk.columns = [f"{c}_lag{L}" for c in cols]
        lag_blocks.append(blk)
    return pd.concat([df] + lag_blocks, axis=1, copy=False)


def score_features(df: pd.DataFrame, feature_cols: list, target_col="target_fwd1"):
    """Ranking po |Spearman|. Pomija cechy zbyt stałe i zbyt krótkie."""
    scores = []
    y_full = df[target_col].astype(float)
    for c in feature_cols:
        try:
            x_full = df[c].astype(float)
            ok = x_full.notna() & y_full.notna()
            n = int(ok.sum())
            if n < MIN_OBS:
                continue
            x = x_full[ok]
            y = y_full[ok]
            if x.nunique(dropna=True) <= 1 or y.nunique(dropna=True) <= 1:
                continue
            rho = x.corr(y, method="spearman")
            if pd.notna(rho):
                scores.append((c, float(abs(rho))))
        except Exception:
            # bezpiecznie pomijamy pojedyncze problemy kolumn
            continue
    scores.sort(key=lambda t: t[1], reverse=True)
    return pd.DataFrame(scores, columns=["feature", "abs_spearman"])


def load_history(max_lines=HIST_WINDOW):
    if not HIST_PATH.exists():
        return []
    rows = []
    for line in HIST_PATH.read_text(encoding="utf-8").splitlines():
        try:
            rows.append(json.loads(line))
        except Exception:
            pass
    return rows[-max_lines:] if rows else []


def save_history(entry: dict):
    HIST_PATH.parent.mkdir(parents=True, exist_ok=True)
    with HIST_PATH.open("a", encoding="utf-8") as f:
        f.write(json.dumps(entry, ensure_ascii=False) + "\n")


def jaccard(a: set, b: set) -> float:
    """|A ∩ B| / |A ∪ B| z ochroną przed dzieleniem przez 0."""
    if not a and not b:
        return 1.0
    return len(a & b) / max(1, len(a | b))


def apply_repeat_penalty(rank_df: pd.DataFrame, hist_rows: list):
    """Obniża punktację cechom często wybieranym w poprzednich tygodniach."""
    if not hist_rows or rank_df.empty:
        return rank_df.copy()
    freq = {}
    for r in hist_rows:
        for c in r.get("feature_columns", []):
            freq[c] = freq.get(c, 0) + 1
    out = rank_df.copy()
    out["repeat_freq"] = out["feature"].map(lambda c: freq.get(c, 0))
    out["score_after_penalty"] = out["abs_spearman"] * (REPEAT_PENALTY ** out["repeat_freq"])
    return out.sort_values("score_after_penalty", ascending=False).reset_index(drop=True)


def diversify_select(adjusted_rank: pd.DataFrame, prev_features: list, topK: int,
                     min_new_ratio=MIN_NEW_RATIO):
    """Wymusza min. udział nowych cech w topK, resztę dopełnia z rankingu."""
    prev_set = set(prev_features or [])
    new_needed = int(np.ceil(topK * min_new_ratio))
    chosen, new_cnt = [], 0
    # Najpierw nowe
    for _, row in adjusted_rank.iterrows():
        f = row["feature"]
        if f not in prev_set and f not in chosen:
            chosen.append(f)
            new_cnt += 1
            if new_cnt >= new_needed:
                break
    # Potem dopełnienie
    if len(chosen) < topK:
        for _, row in adjusted_rank.iterrows():
            f = row["feature"]
            if f not in chosen:
                chosen.append(f)
                if len(chosen) >= topK:
                    break
    return chosen[:topK]


def features_hash(features: list) -> str:
    import hashlib
    return hashlib.sha1(",".join(sorted(features or [])).encode("utf-8")).hexdigest()[:12]

def live_feature_universe(max_lag: int):
    cols = set(LIVE_BASE)
    for L in range(1, max_lag + 1):
        for c in LIVE_BASE:
            cols.add(f"{c}_lag{L}")
    return cols
