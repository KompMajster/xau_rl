# -*- coding: utf-8 -*-
"""
rl/train_ppo.py — nocny (co 8h) trening PPO z:
- preferencją najnowszego speca cech (LIVE → fallback FULL),
- automatyczną reakcją na zmianę spec_id (świeże VecNormalize),
- stabilniejszą walidacją (częstsze eval, więcej epizodów),
- opcjonalnym multi-seed i bogatszym feedbackiem do discovery.

Zależności i struktura są zgodne z repo:
- env_xau.py w root (bezpieczny PYTHONPATH),
- features.csv zbudowane wcześniej,
- models/features_spec_*.json z discovery,
- zapis modelu i normalizacji jak dotąd.
"""

import argparse
import json
import sys
from pathlib import Path
from datetime import datetime
import numpy as np
import pandas as pd
import yaml

from gymnasium.wrappers import TimeLimit
from stable_baselines3 import PPO
from stable_baselines3.common.callbacks import EvalCallback
from stable_baselines3.common.utils import set_random_seed
from stable_baselines3.common.vec_env import (
    DummyVecEnv,
    VecMonitor,
    VecNormalize,
    sync_envs_normalization,
)

# --- PYTHONPATH safety ---
REPO_ROOT = Path(__file__).resolve().parents[1]  # <-- jak w Twojej wersji
if str(REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(REPO_ROOT))

from env_xau import XauTradingEnv  # noqa: E402


# ---------- Helpers: spec & metrics ----------

def _load_json(p: Path) -> dict | None:
    try:
        if p.exists():
            return json.loads(p.read_text(encoding="utf-8"))
    except Exception:
        return None
    return None


def pick_latest_spec() -> tuple[dict | None, str]:
    """
    Preferuj LIVE-safe, ale ostatecznie wybierz po 'generated_at' najnowszy:
    - models/features_spec_live.json
    - models/features_spec.json
    Zwraca (spec_dict, source_name).
    """
    live_p = REPO_ROOT / "models" / "features_spec_live.json"
    full_p = REPO_ROOT / "models" / "features_spec.json"

    live = _load_json(live_p)
    full = _load_json(full_p)

    if live and full:
        gt_live = live.get("generated_at")
        gt_full = full.get("generated_at")
        try:
            dt_live = datetime.fromisoformat(str(gt_live)) if gt_live else None
            dt_full = datetime.fromisoformat(str(gt_full)) if gt_full else None
        except Exception:
            dt_live, dt_full = None, None

        if (dt_live and dt_full and dt_live >= dt_full) or (dt_live and not dt_full):
            return live, "live"
        else:
            return full, "full"
    elif live:
        return live, "live"
    elif full:
        return full, "full"
    else:
        return None, "none"


def intersect_features(spec: dict | None, df_cols: list[str]) -> tuple[list[str], list[str], str, int]:
    """
    Przecięcie kolumn specyfikacji z rzeczywiście dostępnymi w features.csv.
    Zwraca: (used_cols, missing_cols, spec_id, max_lag)
    """
    if not spec:
        return [], [], "", 0

    raw = spec.get("feature_columns", []) or []
    max_lag = int(spec.get("max_lag", 0) or 0)
    used = [c for c in raw if c in df_cols]
    missing = [c for c in raw if c not in df_cols]
    spec_id = str(spec.get("spec_id", ""))

    return used, missing, spec_id, max_lag


def read_last_train_spec_id(metrics_path: Path) -> str | None:
    """
    Odczytaj ostatni spec_id z models/train_metrics.jsonl, aby wykryć zmianę zestawu cech.
    """
    if not metrics_path.exists():
        return None
    try:
        txt = metrics_path.read_text(encoding="utf-8").strip().splitlines()
        for line in reversed(txt):
            try:
                row = json.loads(line)
                sid = row.get("spec_id")
                if sid:
                    return str(sid)
            except Exception:
                continue
    except Exception:
        return None
    return None


# ---------- Main ----------

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--timesteps", type=int, default=1_500_000)
    ap.add_argument("--seed", type=int, default=42)
    ap.add_argument("--seeds", type=str, default="", help="Lista seedów rozdzielona przecinkami, np. '42,1337,2025'")
    ap.add_argument("--eval_freq", type=int, default=50_000)  # częstsza ewaluacja
    ap.add_argument("--n_eval_episodes", type=int, default=3)  # stabilniejsza metryka
    ap.add_argument("--max_train_steps", type=int, default=6000)
    ap.add_argument("--max_eval_steps", type=int, default=3000)
    args = ap.parse_args()

    # --- Config ---
    cfg = yaml.safe_load(open(REPO_ROOT / "config.yaml", "r", encoding="utf-8"))
    FEAT = cfg["files"]["features_csv"]
    MODEL_PATH = Path(cfg["files"]["model_path"])
    VECNORM_PATH = Path(cfg.get("files", {}).get("vecnorm_path", "models/vecnorm_xauusd_m5.pkl"))
    WINDOW = int(cfg.get("window", 128))
    COSTS = cfg["costs"]
    ENV_CFG = cfg.get("env", {}) or {}
    REWARD_MODE = ENV_CFG.get("reward_mode", cfg.get("reward_mode", "pct"))
    FLIP_PENALTY = float(ENV_CFG.get("flip_penalty", cfg.get("flip_penalty", 0.0)))
    TRADE_HOURS = ENV_CFG.get("trade_hours_utc", cfg.get("trade_hours_utc", None))
    MIN_EQ = float(ENV_CFG.get("min_equity", 0.8))

    # --- Dane ---
    df = pd.read_csv(REPO_ROOT / FEAT, parse_dates=["time"])
    df = df.sort_values("time").drop_duplicates("time").reset_index(drop=True)

    # Walidacja = ostatnie 30 dni; trenowanie = reszta (jak dotąd)
    cut_val = df["time"].max() - pd.Timedelta(days=30)
    train_df = df[df["time"] < cut_val].copy()
    val_df = df[df["time"] >= cut_val].copy()

    # --- Spec cech: preferuj najnowszy (LIVE vs FULL), przecięcie z df ---
    spec, source = pick_latest_spec()  # NEW
    used_cols, missing_cols, spec_id, max_lag = intersect_features(spec, list(df.columns))  # NEW

    # Fallback: jeżeli spec nie istnieje lub przecięcie puste → None (użyj wszystkich dostępnych cech jak wcześniej)
    features_spec = used_cols if len(used_cols) > 0 else None

    # Logging diagnostyczny (stdout → trafi do logów daemona)
    print(f"[train_ppo] spec_source={source} spec_id='{spec_id}' used={len(used_cols)} missing={len(missing_cols)} max_lag={max_lag}")

    # --- Reakcja na zmianę spec_id: świeża normalizacja (NEW) ---
    last_spec_id = read_last_train_spec_id(REPO_ROOT / "models" / "train_metrics.jsonl")
    fresh_normalize = (spec_id and last_spec_id and spec_id != last_spec_id)
    if fresh_normalize:
        print(f"[train_ppo] Detected new spec_id vs last training: last='{last_spec_id}' -> current='{spec_id}'. Starting fresh VecNormalize.")

    # --- Seeds do pętli (opcjonalny multi-seed) ---
    seeds_list = []
    if args.seeds.strip():
        try:
            seeds_list = [int(s.strip()) for s in args.seeds.split(",") if s.strip()]
        except Exception:
            seeds_list = []
    if not seeds_list:
        seeds_list = [int(args.seed)]

    # --- Fabryki env ---
    def make_train():
        env = XauTradingEnv(
            train_df,
            window=WINDOW,
            spread_abs=COSTS["spread_abs"],
            commission_rate=COSTS["commission_rate"],
            slippage_k=COSTS["slippage_k"],
            reward_mode=REWARD_MODE,
            use_close_norm=True,
            flip_penalty=FLIP_PENALTY,
            trade_hours_utc=TRADE_HOURS,
            enforce_flat_outside_hours=True,
            features_spec=features_spec,  # przecięcie
            min_equity=MIN_EQ,
        )
        return TimeLimit(env, max_episode_steps=args.max_train_steps)

    def make_eval():
        env = XauTradingEnv(
            val_df,
            window=WINDOW,
            spread_abs=COSTS["spread_abs"],
            commission_rate=COSTS["commission_rate"],
            slippage_k=COSTS["slippage_k"],
            reward_mode=REWARD_MODE,
            use_close_norm=True,
            flip_penalty=0.0,
            trade_hours_utc=TRADE_HOURS,
            enforce_flat_outside_hours=True,
            features_spec=features_spec,  # przecięcie
            min_equity=MIN_EQ,
        )
        return TimeLimit(env, max_episode_steps=args.max_eval_steps)

    # --- Trenowanie w pętli po seedach: wybierz najlepszy model wg EvalCallback ---
    best_score = None
    best_seed = None
    best_model_zip = None
    best_vecnorm_pkl = None

    for idx, sd in enumerate(seeds_list, start=1):
        print(f"[train_ppo] === Seed {sd} ({idx}/{len(seeds_list)}) ===")
        set_random_seed(sd)

        # Train VecEnv — nowa normalizacja (NEW), bez wczytywania starego stanu
        venv_train = DummyVecEnv([make_train])
        venv_train = VecMonitor(venv_train)
        venv_train = VecNormalize(
            venv_train,
            norm_obs=True,
            norm_reward=True,
            clip_obs=10.0,
            clip_reward=10.0,
        )

        # Eval VecEnv — bez treningu, norm_obs=True, reward nie normalizujemy
        venv_eval = DummyVecEnv([make_eval])
        venv_eval = VecMonitor(venv_eval)
        venv_eval = VecNormalize(venv_eval, training=False, norm_obs=True, norm_reward=False)

        # Synchronizacja statystyk (jak w Twojej wersji)
        sync_envs_normalization(venv_eval, venv_train)

        # PPO hiperparametry — zgodne z Twoją wersją; zostawiamy policy_kwargs i LR
        policy_kwargs = dict(net_arch=dict(pi=[128, 128], vf=[128, 128]))
        try:
            import tensorboard as _tb  # noqa: F401
            tb_log_dir = "logs/ppo_gold_m5"
        except Exception:
            tb_log_dir = None

        model = PPO(
            "MlpPolicy",
            venv_train,
            n_steps=4096,
            batch_size=256,
            learning_rate=3e-4,
            ent_coef=0.02,
            policy_kwargs=policy_kwargs,
            seed=sd,
            verbose=1,
            tensorboard_log=tb_log_dir,
        )

        eval_cb = EvalCallback(
            venv_eval,
            best_model_save_path=str(MODEL_PATH.parent),
            log_path="logs/eval",
            eval_freq=max(args.eval_freq, 1),  # częściej
            n_eval_episodes=max(args.n_eval_episodes, 1),  # stabilniej
            deterministic=True,
            render=False,
        )

        model.learn(total_timesteps=args.timesteps // max(1, len(seeds_list)), callback=eval_cb)

        # Zapis modelu i normalizacji dla tego seeda (tymczasowo)
        tmp_model_path = MODEL_PATH.with_name(f"{MODEL_PATH.stem}_seed{sd}{MODEL_PATH.suffix}")
        tmp_vecnorm_path = VECNORM_PATH.with_name(f"{VECNORM_PATH.stem}_seed{sd}{VECNORM_PATH.suffix}")

        MODEL_PATH.parent.mkdir(parents=True, exist_ok=True)
        model.save(str(tmp_model_path))
        venv_train.save(str(tmp_vecnorm_path))
        print(f"[train_ppo] Saved seed {sd}: model={tmp_model_path.name}, vecnorm={tmp_vecnorm_path.name}")

        # Feedback: best_mean_reward z EvalCallback (jak u Ciebie)
        best_mean = getattr(eval_cb, "best_mean_reward", None)
        print(f"[train_ppo] seed={sd} best_mean_reward={best_mean}")

        # Wybór najlepszego seeda
        cur_score = float(best_mean) if best_mean is not None else -np.inf
        if (best_score is None) or (cur_score > best_score):
            best_score = cur_score
            best_seed = sd
            best_model_zip = tmp_model_path
            best_vecnorm_pkl = tmp_vecnorm_path

        # Zwolnij envy przed następnym seedem
        try:
            venv_train.close()
            venv_eval.close()
        except Exception:
            pass

    # --- Zapisy finalne: najlepszy seed trafia pod docelowe ścieżki ---
    if best_model_zip and best_model_zip.exists():
        best_bytes = best_model_zip.read_bytes()
        MODEL_PATH.write_bytes(best_bytes)
    if best_vecnorm_pkl and best_vecnorm_pkl.exists():
        best_bytes = best_vecnorm_pkl.read_bytes()
        VECNORM_PATH.write_bytes(best_bytes)

    print(f"[train_ppo] Saved BEST model: {MODEL_PATH} (seed={best_seed}, score={best_score})")
    print(f"[train_ppo] Saved BEST VecNormalize: {VECNORM_PATH}")

    # --- Zapis feedbacku dla discovery/ops (bogatszy) ---
    try:
        metrics = {
            "ts": pd.Timestamp.utcnow().isoformat(),
            "mode": "train_ppo",
            "spec_id": spec_id or "",
            "spec_source": source,
            "used_features": len(used_cols),
            "missing_features": len(missing_cols),
            "best_mean_reward": float(best_score) if best_score is not None else None,
            "best_seed": int(best_seed) if best_seed is not None else None,
            "timesteps": int(args.timesteps),
            "eval_freq": int(args.eval_freq),
            "n_eval_episodes": int(args.n_eval_episodes),
        }
        Path(REPO_ROOT / "models").mkdir(parents=True, exist_ok=True)
        with open(REPO_ROOT / "models" / "train_metrics.jsonl", "a", encoding="utf-8") as f:
            f.write(json.dumps(metrics, ensure_ascii=False) + "\n")
        print("[train_ppo] feedback saved -> models/train_metrics.jsonl")
    except Exception as _e:
        print(f"[train_ppo][warn] feedback not saved: {_e}")


if __name__ == "__main__":
    try:
        main()
    except Exception:
        import traceback
        print("\n[train_ppo][FATAL] Uncaught exception:\n", file=sys.stderr, flush=True)
        traceback.print_exc()
        sys.exit(1)
